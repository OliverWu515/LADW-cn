\section*{Chapter 1}
\begin{enumerate}[leftmargin=*]
    \item \textbf{Page 6, Definition 2.1, last paragraph}: 
          $x_{1}\mathbf{v}_{1}+x_{2}\mathbf{v}_{2}+\ldots+x_{m}\mathbf{v}_{n}=\mathbf{v}$ should be 
          $x_{1}\mathbf{v}_{1}+x_{2}\mathbf{v}_{2}+\ldots+x_{n}\mathbf{v}_{n}=\mathbf{v}$.
    
    \item \textbf{Page 6, Example 2.2, last displayed formula}: 
          $\mathbf{v}=x_1\mathbf{e}_1+x_2\mathbf{e}_2+\ldots x_n\mathbf{e}_n=\sum_{k=1}^nx_k\mathbf{e}_k$ should be 
          $\mathbf{v}=x_1\mathbf{e}_1+x_2\mathbf{e}_2+\ldots +x_n\mathbf{e}_n=\sum_{k=1}^nx_k\mathbf{e}_k$.
    
    \item \textbf{Page 8, Definition of linear combination, first paragraph}: 
          $\alpha_{k}=\mathbf{0}\ \forall k$ should be $\alpha_{k}=0\ \forall k$.
    
    \item \textbf{Page 12, ``Reflection" example}: 
          The note ``see the fig" should be included in the preceding example.
    
    \item \textbf{Page 20, First displayed formula}: 
          $\forall {x}\in\mathbb{F}^r$ should be $\forall \mathbf{x}\in\mathbb{F}^r$.
    
    \item \textbf{Page 27, Last line}: 
          $\sum_{k=0}^{n}a_{k}t^{k},\alpha_{k}\in\mathbb{F}$ should be 
          $\sum_{k=0}^{n}a_{k}t^{k},a_{k}\in\mathbb{F}$.
    
    \item \textbf{Page 28, Item 3}: 
          ``is isomorphic to $\mathbb{R}^6$" should be ``is isomorphic to $\mathbb{F}^6$".
    
    \item \textbf{Page 36, Footnote 3}: 
          $\mathbb{R}^2$ should be $\mathbb{R}^3$, and $0$ should be $\mathbf{0}$.
\end{enumerate}

\end{enumerate}

\section*{Chapter 3}
\begin{enumerate}[leftmargin=*]
    \item \textbf{Page 77, First displayed formula}: 
          ``the distance from $v_j$ to $\operatorname{span}\{\mathbf{v}_r : r \ne j\}$" should be 
          ``the distance from $\mathbf{v}_j$ to $\operatorname{span}\{\mathbf{v}_r : r \ne j\}$".
    
    \item \textbf{Page 79, First displayed formula}: 
          Remove the extra equals sign in the first line.
    
    \item \textbf{Pages 82-83, Matrix notation}: 
          $A=\{a_{j,k}\}_{j,j=1}^{n}$ should be $A=\{a_{j,k}\}_{j,k=1}^{n}$.
    
    \item \textbf{Page 88, Second displayed formula and last line of following paragraph}: 
          $D(\mathbf{e}_{j_1}.\mathbf{e}_{j_2},\ldots\mathbf{e}_{j_n})$ should be 
          $D(\mathbf{e}_{j_1},\mathbf{e}_{j_2},\ldots,\mathbf{e}_{j_n})$.
    
    \item \textbf{Page 91, Last line}: 
          ``interchanging rows 2 and 3" should be ``interchanging columns 2 and 3".
    
    \item \textbf{Page 94, Second line}: 
          ``multiplying $j$th row of $A$ by $j$th column of $A$" should be 
          ``multiplying $j$th row of $A$ by $j$th column of $C^T$".
    
    \item \textbf{Page 97, Proof of Corollary 6.2, second paragraph}: 
          All instances of $k$ should be $r$.
\end{enumerate}

    \item \textbf{Page 194, Fourth paragraph, third line}:

    \item \textbf{Page 197, Exercise 6.4, first line}:
