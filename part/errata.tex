\chapter*{翻译版中更正的原书笔误}
\markboth{翻译版中更正的原书笔误}{翻译版中更正的原书笔误}

\section*{第1章}

{\bf 原书第6页（翻译第5页）定义2.1最后一段}：$x_{1}\mathbf{v}_{1}+x_{2}\mathbf{v}_{2}+\ldots+x_{m}\mathbf{v}_{n}=\mathbf{v}$改为$x_{1}\mathbf{v}_{1}+x_{2}\mathbf{v}_{2}+\ldots+x_{n}\mathbf{v}_{n}=\mathbf{v}$

{\bf 原书第6页例2.2最后一个行间公式（翻译第6页）}：$\mathbf{v}=x_1\mathbf{e}_1+x_2\mathbf{e}_2+\ldots x_n\mathbf{e}_n=\sum_{k=1}^nx_k\mathbf{e}_k$改为$\mathbf{v}=x_1\mathbf{e}_1+x_2\mathbf{e}_2+\ldots +x_n\mathbf{e}_n=\sum_{k=1}^nx_k\mathbf{e}_k$

{\bf 原书第8页线性组合定义的第一段（翻译第8页）}：$\alpha_{k}=\mathbf{0}\ \forall k$改为$\alpha_{k}=0\ \forall k$

{\bf 原书第12页（翻译第11页）“反射”的例子}：see the fig（见图）应该在上面的例子中。

{\bf 原书第20页第一个行间公式（翻译第19页）}：$\forall {x}\in\mathbb{F}^r$应为$\forall \mathbf{x}\in\mathbb{F}^r$

{\bf 原书第27页最后一行（翻译第25页）}：$\sum_{k=0}^{n}a_{k}t^{k},\alpha_{k}\in\mathbb{F}$应为$\sum_{k=0}^{n}\alpha_{k}t^{k},\alpha_{k}\in\mathbb{F}$

{\bf 原书第28页第3点（翻译第26页）}：is isomorphic to $\mathbb{R}^6$应为is isomorphic to $\mathbb{F}^6$

{\bf 原书第36页脚注3（翻译第33页）}：$\mathbb{R}^2$应为$\mathbb{R}^3$，$0$应为$\mathbf{0}$

\section*{第2章}

{\bf 原书第54页命题5.1下一段第四行（翻译第49页）}：$n=\dim E$应为$n=\dim V$

{\bf 原书第66页最后一段（翻译第61页第一段）}：$\mathbf{v}_1,\mathbf{v}_2,\ldots,\mathbf{v}_r,\mathbf{v}_k\in\mathbb{F}^n$应为$\mathbf{v}_1,\mathbf{v}_2,\ldots,\mathbf{v}_r\in\mathbb{F}^n$，$\mathbb{R}^n$应为$\mathbb{F}^n$

\section*{第3章}

{\bf 原书77页（翻译第70页）第一个行间公式}：the distance from $v_j$ to $\operatorname{span}\{\mathbf{v}_r : r \ne j\}$ 应为 the distance from $\mathbf{v}_j$ to $\operatorname{span}\{\mathbf{v}_r : r \ne j\}$.

{\bf 原书79页（翻译第71页）第一个行间公式}：第一行多了一个等号

{\bf 原书82、83页（翻译第74、75页）矩阵记号}：$A=\{a_{j,k}\}_{j,j=1}^{n}$应为$A=\{a_{j,k}\}_{j,k=1}^{n}$

{\bf 原书88页（翻译第79页）第二个行间公式及下面一段最后一行}：$D(\mathbf{e}_{j_1}.\mathbf{e}_{j_2},\ldots\mathbf{e}_{j_n})$应为$D(\mathbf{e}_{j_1},\mathbf{e}_{j_2},\ldots,\mathbf{e}_{j_n})$

{\bf 原书91页（翻译第83页）最后一行}：interchanging rows 2 and 3（交换第2、3行）应为interchanging columns 2 and 3（交换第2、3列）。

{\bf 原书94页（翻译第85页）第二行}：multiplying $j$th row of $A$ by $j$th column of $A$应为multiplying $j$th row of $A$ by $j$th column of $C^T$。

{\bf 原书97页（翻译第88页）推论6.2证明第2段}：所有 $k$ 均应为$r$

\section*{第4章}

{\bf 原书99页（翻译第90页）脚注}：$e^{tA}:=\sum_{k=0}^{\infty}t^{k}A^{n}/k!$应为$e^{tA}:=\sum_{k=0}^{\infty}t^{k}A^{k}/k!$

{\bf 原书106页（翻译第96页）倒数第二段}：$D=S^{-1}AS=[I]_\mathcal{BS}A[I]_\mathcal{SB}]$应为$D=S^{-1}AS=[I]_\mathcal{BS}A[I]_\mathcal{SB}$。为与第 2 章第 8 节描述一致，下标的逗号均删去。

{\bf 原书113页（翻译第102页）2.7.2}：$\ker (A-\lambda T)$应为$\ker (A-\lambda I)$

\section*{第5章}

{\bf 原书127页（翻译第115页）第一行}：$\|\mathbf{v}_k\|\ne\mathbf{0}$应为$\|\mathbf{v}_k\|\ne0$

{\bf 原书131页（翻译第119页）第二行}：$k=1,2,\dots,n$应为$k=1,2,\dots,r$

{\bf 原书137页（翻译第124页）第二个行间公式上方两行}：$A\mathbf{x}\perp\rg A$应为$\mathbf{b}-A\mathbf{x}\perp\rg A$

{\bf 原书144页（翻译第132页）定理5.1证明倒数第二行}：${x}\in(\rg A)^\perp$应为$\mathbf{x}\in(\rg A)^\perp$

{\bf 原书153页（翻译第140页）定理7.1证明最后一个行间公式}：$T(\mathbf{x}-\alpha\mathbf{y})=T(\mathbf{x})+\alpha T(\mathbf{y})$应为$T(\mathbf{x}+\alpha\mathbf{y})=T(\mathbf{x})+\alpha T(\mathbf{y})$

\section*{第6章}

{\bf 原书165页定理1.2倒数第二行}：$T=UTU^*=UTU^T$应为$A=UTU^*=UTU^T$

{\bf 原书167页命题2.3证明第一行}：定理1.1应为定理2.1

{\bf 原书171页习题2.10}：where $\alpha$ is not a multiple of $\pi$ 应为 where $\alpha\ne n\pi,n\in \mathbb{Z}$ 

{\bf 原书173页3.2节标题}：Singular values 应为 Polar decomposition

{\bf 原书173页定理3.5证明第五行}：$U$ 应为 $U_0$

{\bf 原书177页定义3.9第三行}：$W\in M_{n\times n}^{\mathbb{F}},V\in M_{m\times m}^{\mathbb{F}}$应为$V\in M_{n\times n}^{\mathbb{F}},W\in M_{m\times m}^{\mathbb{F}}$

{\bf 原书179页第4节第1段倒数第二行}：(3.5) 应为 (3.4)

{\bf 原书180页倒数第二段第三行}：$\sigma_1,\sigma_2$ and $\sigma_2$ 应为 $\sigma_1,\sigma_2$ and $\sigma_3$

{\bf 原书182页第二段第二行}：(3.5) 应为 (3.4)，$A=W\Sigma V$ 应为 $A=W\Sigma V^*$
